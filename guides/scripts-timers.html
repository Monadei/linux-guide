systemd timers

Задачи по расписанию.

Сервис + таймер

# ~/.config/systemd/user/backup.service
[Unit]
Description=Backup

[Service]
Type=oneshot
ExecStart=/home/monadei/backup.sh
# ~/.config/systemd/user/backup.timer
[Unit]
Description=Daily backup

[Timer]
OnCalendar=daily
Persistent=true

[Install]
WantedBy=timers.target

Активация

systemctl --user daemon-reload
systemctl --user enable --now backup.timer
systemctl --user list-timers

Синтаксис OnCalendar

daily              # каждый день 00:00
hourly             # каждый час
weekly             # раз в неделю
Mon..Fri 09:00     # по будням в 9
*-*-1 03:00        # 1-го числа
*:0/15             # каждые 15 минут
