Команды для системы

systemctl, journalctl, dmesg, htop — управление системой.

systemctl — сервисы

systemctl status nginx                  # статус
systemctl start nginx                   # запустить
systemctl stop nginx                    # остановить
systemctl restart nginx                 # перезапустить
systemctl reload nginx                  # перечитать конфиг
systemctl enable nginx                  # автозапуск
systemctl disable nginx                 # убрать автозапуск
systemctl is-enabled nginx
systemctl list-units --failed           # что упало
systemctl list-units --type=service
systemctl list-timers                   # таймеры

journalctl — логи systemd

journalctl                              # все логи
journalctl -b                           # с последней загрузки
journalctl -b -1                        # предыдущая загрузка
journalctl -xe                          # ошибки
journalctl -f                           # следить
journalctl -n 50                        # последние 50 строк
journalctl -p err                       # только ошибки
journalctl --since "1 hour ago"
journalctl --since today
journalctl -u nginx                     # по юниту
journalctl -u nginx -f
journalctl --disk-usage
sudo journalctl --vacuum-size=500M

dmesg — сообщения ядра

dmesg                                   # все
dmesg | tail -50                        # последние
dmesg | grep -i error                   # ошибки
dmesg -w                                # следить
dmesg -l err                            # только ошибки
dmesg | grep -i usb
dmesg | grep -i wifi

htop, btop, top

top                                     # классика
htop                                    # удобнее
btop                                    # красивее

# В htop:
F1 — помощь
F3 — поиск
F4 — фильтр
F5 — дерево
F6 — сортировка
F9 — kill
F10 — выход

Ресурсы

free -h                                 # память
df -h                                   # место на дисках
du -sh папка/                           # размер папки
uptime                                  # время работы
w                                       # кто в системе
who                                     # то же
last                                    # история входов

# Аппаратное
lscpu                                   # процессор
lsblk                                   # диски
lspci                                   # PCI-устройства
lsusb                                   # USB-устройства
lsmod                                   # модули ядра
uname -a                                # инфо о ядре

Загрузка

systemd-analyze                         # время загрузки
systemd-analyze blame                   # что долго грузится
systemd-analyze critical-chain          # цепочка
systemd-analyze plot > boot.svg         # график

Питание

systemctl suspend                        # сон
systemctl hibernate                      # гибернация
systemctl reboot                         # перезагрузка
systemctl poweroff                       # выключение

# Узнать, что блокирует сон
systemd-inhibit --list
