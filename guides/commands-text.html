Команды для текста

grep, sed, awk, cut, sort, uniq — работа с текстом.

grep — поиск

grep "текст" file.txt
grep -r "текст" .              # рекурсивно
grep -i "текст" file.txt       # игнорировать регистр
grep -v "текст" file.txt       # инвертировать
grep -c "текст" file.txt       # количество совпадений
grep -n "текст" file.txt       # с номерами строк
grep -E "regex" file.txt       # расширенные regex
grep -A 3 "текст" file.txt     # 3 строки после
grep -B 3 "текст" file.txt     # 3 строки до

sed — замена

sed 's/старое/новое/' file.txt          # первое в строке
sed 's/старое/новое/g' file.txt         # все вхождения
sed -i 's/старое/новое/g' file.txt      # прямо в файле
sed -n '5,10p' file.txt                  # строки 5-10
sed '/шаблон/d' file.txt                # удалить строки
sed '1d' file.txt                        # удалить первую
sed '$d' file.txt                        # удалить последнюю
sed 's/^/  /' file.txt                  # отступ всем строкам

awk — обработка колонок

awk '{print $1}' file.txt               # первая колонка
awk -F: '{print $1}' /etc/passwd        # с разделителем :
awk '{print $1, $3}' file.txt           # две колонки
awk '$3 > 100' file.txt                 # фильтр
awk 'NR > 1' file.txt                   # пропустить первую
awk '{sum += $1} END {print sum}' file.txt  # сумма
awk 'length > 80' file.txt              # строки > 80 символов

cut — вырезать колонки

cut -d: -f1 /etc/passwd                 # 1-я колонка
cut -d: -f1,3 /etc/passwd               # 1-я и 3-я
cut -c1-10 file.txt                     # символы 1-10
cut -c1,5,10 file.txt                   # конкретные символы

sort и uniq

sort file.txt                           # алфавитно
sort -n file.txt                        # численно
sort -r file.txt                        # обратный
sort -u file.txt                        # без дублей
sort -k2 file.txt                       # по 2-й колонке

uniq file.txt                           # удалить подряд идущие дубли
uniq -c file.txt                        # с подсчётом
uniq -d file.txt                        # только дубли
sort file.txt | uniq -c | sort -rn      # топ по частоте

wc — подсчёт

wc file.txt                             # строки, слова, символы
wc -l file.txt                          # только строки
wc -w file.txt                          # только слова
wc -c file.txt                          # только байты

Полезные сочетания

# Топ-10 самых частых слов в файле
cat file.txt | tr ' ' '\n' | sort | uniq -c | sort -rn | head -10

# Найти все IP в логе
grep -oE '\b[0-9]{1,3}(\.[0-9]{1,3}){3}\b' access.log | sort -u

# Заменить во всех файлах
grep -rl "старое" . | xargs sed -i 's/старое/новое/g'

# Удалить пустые строки
sed '/^$/d' file.txt

# Убрать дубли строк, сохранив порядок
awk '!seen[$0]++' file.txt
