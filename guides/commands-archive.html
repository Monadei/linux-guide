Команды для архивов

tar, gzip, zip, 7z, rsync — архивы и синхронизация.

tar — основной инструмент

# Создать архив
tar -czf archive.tar.gz папка/          # gzip
tar -cjf archive.tar.bz2 папка/         # bzip2
tar -cJf archive.tar.xz папка/          # xz

# Распаковать
tar -xzf archive.tar.gz
tar -xjf archive.tar.bz2
tar -xJf archive.tar.xz

# Посмотреть содержимое без распаковки
tar -tzf archive.tar.gz

# Распаковать в другую папку
tar -xzf archive.tar.gz -C /path/to/

# Исключить файлы
tar -czf backup.tar.gz --exclude='*.log' папка/

gzip, bzip2, xz

gzip file.txt                    # file.txt → file.txt.gz
gunzip file.txt.gz
bzip2 file.txt
bunzip2 file.txt.bz2
xz file.txt
unxz file.txt.xz

zip и unzip

zip archive.zip file1 file2
zip -r archive.zip папка/
zip -e archive.zip file.txt      # с паролем

unzip archive.zip
unzip archive.zip -d /path/
unzip -l archive.zip             # список
unzip -p archive.zip file.txt    # в stdout

7z — максимальное сжатие

sudo apt install p7zip-full

7z a archive.7z папка/
7z x archive.7z
7z l archive.7z
7z a -p archive.7z папка/        # с паролем

rsync — синхронизация

# Основное
rsync -avh источник/ назначение/
rsync -avh --progress источник/ назначение/
rsync -avh --delete источник/ назначение/    # удалить лишнее

# На удалённый сервер
rsync -avh -e ssh ~/Documents/ user@server:/backup/

# С сервера
rsync -avh user@server:/data/ ~/local/

# Исключить файлы
rsync -avh --exclude='*.log' источник/ назначение/

# Только список изменений (dry-run)
rsync -avhn источник/ назначение/

Флаги rsync

  • -a — архивный режим (права, время, симлинки)
  • -v — verbose
  • -h — human-readable
  • -z — сжатие при передаче
  • -P — прогресс + продолжить
  • --delete — удалять файлы, которых нет в источнике
  • --exclude — исключить по шаблону
  • -n — dry-run (не делать, только показать)
