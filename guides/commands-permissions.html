Команды для прав

chmod, chown, umask, ACL, sudo — управление правами.

chmod — права доступа

# Числовая запись
chmod 755 file          # rwxr-xr-x
chmod 644 file          # rw-r--r--
chmod 600 file          # rw------- (SSH-ключи)
chmod 700 папка/        # rwx------

# Буквенная запись
chmod +x file           # добавить выполнение
chmod -x file           # убрать выполнение
chmod u+w file          # владельцу — запись
chmod g+r file          # группе — чтение
chmod o-r file          # остальным — убрать чтение

# Рекурсивно
chmod -R 755 папка/

chown — владелец

sudo chown user file
sudo chown user:group file
sudo chown -R user:group папка/
sudo chown --reference=file1 file2      # как у file1

umask — маска прав

Определяет права по умолчанию для новых файлов.

umask                   # показать (обычно 022)
umask 077               # новые файлы только для меня

# В ~/.bashrc или ~/.profile:
umask 022

При umask 022: новые файлы получают 644, папки — 755.

ACL — расширенные права

# Установка
sudo apt install acl

# Дать доступ конкретному пользователю
sudo setfacl -m u:monadei:rwx /var/www/

# Показать ACL
getfacl /var/www/

# Удалить ACL
setfacl -b /var/www/

# По умолчанию для новых файлов в папке
setfacl -d -m u:monadei:rwx /var/www/

sudo и sudoers

sudo команда                        # выполнить от root
sudo -i                             # root-shell
sudo -u monadei команда             # от имени другого
sudo -l                             # что я могу

# Редактирование sudoers
sudo visudo

# Примеры правил в /etc/sudoers:
monadei ALL=(ALL) ALL               # полный sudo с паролем
monadei ALL=(ALL) NOPASSWD: ALL     # без пароля
monadei ALL=(ALL) /usr/bin/apt      # только apt
%admins ALL=(ALL) ALL               # вся группа

Специальные биты

chmod 4755 file         # SUID — выполнить от владельца
chmod 2755 file         # SGID — выполнить от группы
chmod 1777 папка/       # Sticky — удалять только свои файлы

# Найти все SUID-файлы в системе
sudo find / -perm -4000 -type f 2>/dev/null
