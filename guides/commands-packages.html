Команды для пакетов

apt, dnf, pacman, zypper — шпаргалка по пакетным менеджерам.

Сравнительная таблица

apt — подробно

sudo apt update                         # обновить индексы
sudo apt upgrade                        # обновить пакеты
sudo apt full-upgrade                   # с удалением конфликтов
sudo apt install имя
sudo apt remove имя
sudo apt purge имя                      # с конфигами
sudo apt autoremove                     # убрать ненужное
sudo apt search запрос
apt show имя
apt list --installed
sudo apt install --reinstall имя
sudo apt --fix-broken install

dnf — подробно

sudo dnf upgrade --refresh
sudo dnf install имя
sudo dnf remove имя
sudo dnf search запрос
dnf info имя
dnf list installed
sudo dnf clean all
dnf history
sudo dnf history undo ID

pacman — подробно

sudo pacman -Syu                        # обновить всё
sudo pacman -S имя                      # установить
sudo pacman -R имя                      # удалить
sudo pacman -Rns имя                    # с зависимостями
pacman -Ss запрос                       # поиск
pacman -Si имя                          # инфо
pacman -Q                               # установленные
sudo pacman -Sc                         # очистить кеш
yay -S имя                              # AUR
