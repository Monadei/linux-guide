Установка Arch Linux

Пошагово: от загрузки ISO до первого входа в систему.

Что нужно

  • Флешка 8+ ГБ
  • ISO Arch с archlinux.org
  • Интернет
  • 30–60 минут
⚠️ Все данные на диске будут уничтожены. Сделай бэкап.

1. Загрузка с флешки

# Проверь интернет
ping -c 3 archlinux.org

# Если Wi-Fi
iwctl
  station wlan0 scan
  station wlan0 get-networks
  station wlan0 connect "Имя сети"
  exit

2. Разметка диска

# Смотрим диски
lsblk

# Размечаем (замени /dev/sda на свой)
fdisk /dev/sda
# n — новый раздел
# 1 — EFI, 512M, тип EFI
# 2 — root, всё остальное

# Форматируем
mkfs.fat -F32 /dev/sda1
mkfs.ext4 /dev/sda2

# Монтируем
mount /dev/sda2 /mnt
mount --mkdir /dev/sda1 /mnt/boot

3. Установка базовой системы

# Ставим базовые пакеты
pacstrap -K /mnt base linux linux-firmware vim sudo networkmanager

# Генерируем fstab
genfstab -U /mnt >> /mnt/etc/fstab

# Заходим в систему
arch-chroot /mnt

4. Базовая настройка

# Часовой пояс
ln -sf /usr/share/zoneinfo/Europe/Moscow /etc/localtime
hwclock --systohc

# Локаль
echo "en_US.UTF-8 UTF-8" >> /etc/locale.gen
echo "ru_RU.UTF-8 UTF-8" >> /etc/locale.gen
locale-gen
echo "LANG=ru_RU.UTF-8" > /etc/locale.conf

# Имя хоста
echo "arch-pc" > /etc/hostname

# Пароль root
passwd

5. Загрузчик GRUB

# Для UEFI
pacman -S grub efibootmgr
grub-install --target=x86_64-efi --efi-directory=/boot --bootloader-id=GRUB
grub-mkconfig -o /boot/grub/grub.cfg

6. Пользователь и сеть

# Создаём пользователя
useradd -m -G wheel -s /bin/bash monadei
passwd monadei

# Даём sudo
EDITOR=vim visudo
# Раскомментируй: %wheel ALL=(ALL:ALL) ALL

# Включаем сеть
systemctl enable NetworkManager

# Выходим и перезагружаемся
exit
umount -R /mnt
reboot
